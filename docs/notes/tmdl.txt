				   Pas de relation entre DIM_CALENDRIER et dprod : le REMOVEFILTERS reste
				   indispensable pour detacher le calendrier de la fact de flux. */
				VAR Date_Debut = MIN('DIM_CALENDRIER'[DATE])
				VAR Date_Fin   = MAX('DIM_CALENDRIER'[DATE])
				RETURN
				CALCULATE(
				    COUNTROWS('DIM_CONTRAT'),
				    REMOVEFILTERS('DIM_CALENDRIER'),
				    'DIM_CONTRAT'[DPROD] >= Date_Debut,
				    'DIM_CONTRAT'[DPROD] <= Date_Fin
				)

			displayFolder: 01. Ouvertures

		/// CONSEQUENCE A CONNAITRE : sur un mois, P-1 est le mois PRECEDENT, pas le meme mois un an

		/// decembre n est pas un ecart de performance pure.

				VAR Date_Debut_Sel = MIN('DIM_CALENDRIER'[DATE])

				VAR Duree_Jours    = DATEDIFF(Date_Debut_Sel, Date_Fin_Sel, DAY) + 1

				VAR Date_Fin_P1    = Date_Fin_Sel - Duree_Jours

				CALCULATE(
				    [% Taux d'Activité (12 mois)],
				    REMOVEFILTERS('DIM_CALENDRIER'),
